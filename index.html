// 1. Initialize notes array from storage or empty let notes = JSON.parse(localStorage.getItem('flat_notes')) || []; // 2. Save notes function function saveNotes() { localStorage.setItem('flat_notes', JSON.stringify(notes)); } // 3. Add a new note function addNote(title, content) { const newNote = { id: Date.now().toString(), title: title, content: content, updatedAt: new Date().toISOString() }; notes.push(newNote); saveNotes(); renderNotes(); } // 4. Render notes to the DOM function renderNotes() { const container = document.getElementById('notes-container'); container.innerHTML = ''; notes.forEach(note => { const el = document.createElement('div'); el.className = 'note-item'; el.innerHTML = `

${note.title}

${note.content}

`; container.appendChild(el); }); } // 5. Delete a note function deleteNote(id) { notes = notes.filter(note => note.id !== id); saveNotes(); renderNotes(); }
